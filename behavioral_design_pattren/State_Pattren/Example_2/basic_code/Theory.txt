class TrafficLight {

    State redState;
    State greenState;
    State yellowState;

    private State currentState;

    public TrafficLight() {
        redState = new RedState();
        greenState = new GreenState();
        yellowState = new YellowState();

        currentState = redState; // initial
    }

    public void setState(State state) {
        this.currentState = state;
    }

    public State getRedState() {
        return redState;
    }

    public State getGreenState() {
        return greenState;
    }

    public State getYellowState() {
        return yellowState;
    }

    public void next() {
        currentState.next(this);
    }

    public void showSignal() {
        currentState.showSignal();
    }
}


class Red implements State{
  public void next(TrafficLisght tf)
  {
    tf.setState(tf.getGreenState());
  }

  public void show()
  {
    system.out.println("red");
  }
}
