public class Remote{
  Command cmd;
  Stack<Command> history;
  Remote(Command cmd)
  {
    this.cmd=cmd;
    history=new Stack<>();
  }
  public void setCommand(Command cmd)
  {
    this.cmd=cmd;
  }
  public void click()
  {
    cmd.execute();
  }

  public void undo()
  {
    if(!history.isEmpty())
    {
      history.pop().undo();
      return;
    }
    System.out.println("first do something to undo");
  }
}
